import shutil, re

# ============================
# FIX 1: claude_service.py
# Restaurar backup + Anthropic -> AsyncAnthropic
# + await em messages.create APENAS dentro de async def
# ============================
path1 = "/opt/agent-financeiro-aisatec/app/services/claude_service.py"
bak1 = path1 + ".bak_pre_async"

# Restaurar de backup (limpar fix quebrado)
shutil.copy2(bak1, path1)
print("Restaurado de backup: claude_service.py")

with open(path1, "r") as f:
    lines = f.readlines()

ch1 = 0

# Fix 1a: Adicionar AsyncAnthropic ao import
for i, line in enumerate(lines):
    if "from anthropic import Anthropic" in line and "AsyncAnthropic" not in line:
        lines[i] = line.replace(
            "from anthropic import Anthropic",
            "from anthropic import AsyncAnthropic, Anthropic"
        )
        ch1 += 1
        print(f"OK 1a: import AsyncAnthropic (linha {i+1})")
        break

# Fix 1b: self.client = Anthropic() -> AsyncAnthropic()
for i, line in enumerate(lines):
    if "self.client = Anthropic(" in line and "AsyncAnthropic" not in line:
        lines[i] = line.replace("self.client = Anthropic(", "self.client = AsyncAnthropic(")
        ch1 += 1
        print(f"OK 1b: AsyncAnthropic() no __init__ (linha {i+1})")
        break

# Fix 1c: Adicionar await em TODOS os self.client.messages.create
# MAS somente se estiver dentro de uma funcao async def
# (busca para tras ate achar def/async def/class)
for i, line in enumerate(lines):
    if "self.client.messages.create" in line and "await " not in line:
        # Escanear para tras para achar a funcao que contem esta linha
        is_async = False
        for j in range(i - 1, -1, -1):
            stripped_j = lines[j].lstrip()
            if re.match(r'async\s+def\s+', stripped_j):
                is_async = True
                break
            elif re.match(r'def\s+', stripped_j):
                # def sync - NAO adicionar await
                break
            elif re.match(r'class\s+', stripped_j):
                # Chegou na classe sem achar def
                break

        if is_async:
            lines[i] = line.replace(
                "self.client.messages.create",
                "await self.client.messages.create"
            )
            ch1 += 1
            print(f"OK 1c: await adicionado (linha {i+1})")
        else:
            print(f"SKIP: linha {i+1} NAO esta em async def - sem await")

with open(path1, "w") as f:
    f.writelines(lines)

print(f"\nclaude_service.py: {ch1} alteracoes\n")

# ============================
# FIX 2: autentique_service.py
# Restaurar backup + envolver execute() com asyncio.to_thread
# (asyncio ja importado na linha 43 - nao precisa trocar transport)
# ============================
path2 = "/opt/agent-financeiro-aisatec/app/services/autentique_service.py"
bak2 = path2 + ".bak_pre_async"

shutil.copy2(bak2, path2)
print("Restaurado de backup: autentique_service.py")

with open(path2, "r") as f:
    c2 = f.read()

ch2 = 0

# Fix 2a: create_document execute (linha 260)
old = "            result = self.client.execute(mutation, variable_values=document_input)"
new = "            result = await asyncio.to_thread(self.client.execute, mutation, variable_values=document_input)"
if old in c2:
    c2 = c2.replace(old, new, 1)
    ch2 += 1
    print("OK 2a: create_document -> asyncio.to_thread")
else:
    print("WARN 2a: pattern nao encontrado")

# Fix 2b: get_document_status execute (linha 347)
old = '            result = self.client.execute(query, variable_values={"id": document_id})'
new = '            result = await asyncio.to_thread(self.client.execute, query, variable_values={"id": document_id})'
if old in c2:
    c2 = c2.replace(old, new, 1)
    ch2 += 1
    print("OK 2b: get_document_status -> asyncio.to_thread")
else:
    print("WARN 2b: pattern nao encontrado")

# Fix 2c: list_documents execute (linhas 419-422 - multiline)
old = '''            result = self.client.execute(
                query,
                variable_values={"limit": limit, "page": page}
            )'''
new = '''            result = await asyncio.to_thread(
                self.client.execute,
                query,
                variable_values={"limit": limit, "page": page}
            )'''
if old in c2:
    c2 = c2.replace(old, new, 1)
    ch2 += 1
    print("OK 2c: list_documents -> asyncio.to_thread")
else:
    print("WARN 2c: pattern multiline nao encontrado")

with open(path2, "w") as f:
    f.write(c2)

print(f"\nautentique_service.py: {ch2} alteracoes")
print(f"\nTOTAL: {ch1 + ch2} alteracoes")
PYEOF

echo ""

grep -n "AsyncAnthropic\|await.*messages.create" /opt/agent-financeiro-aisatec/app/services/claude_service.py
echo ""

grep -n "asyncio.to_thread" /opt/agent-financeiro-aisatec/app/services/autentique_service.py

systemctl restart agent-financeiro
sleep 3
