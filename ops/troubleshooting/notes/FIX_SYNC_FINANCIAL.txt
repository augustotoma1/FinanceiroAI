                - currency: Currency code (default: BRL)

                - due_date: Payment due date
                - paid_at: Payment date (if paid)

    Returns:

new_sync_fn = """async def _sync_single_transaction(
    db: Session,
    transaction_data: Dict[str, Any]
) -> Tuple[bool, bool]:
    \"\"\"
    Synchronize a single financial transaction (Conta Azul v2 API format)
    to a local FinancialRecord in the database.

    Handles field mapping from Conta Azul v2 API format:
        - id -> conta_azul_id
        - valor / valor_total -> amount
        - status -> status (mapped to lowercase)
        - data_vencimento -> due_date
        - descricao / observacao -> description
        - pessoa.id -> customer lookup
        - _sync_type -> transaction_type (receivable/payable)

    Returns:
        Tuple of (created: bool, updated: bool)
    \"\"\"
    # Extract conta_azul_id (try multiple field names)
    conta_azul_id = transaction_data.get("id") or transaction_data.get("uuid")
    if not conta_azul_id:
        raise FinancialSyncDataError("Transaction missing required 'id' field")

    # Extract customer/supplier from nested 'pessoa' object or flat fields
    pessoa = transaction_data.get("pessoa") or {}
    conta_azul_customer_id = None
    if isinstance(pessoa, dict):
        conta_azul_customer_id = pessoa.get("id")
    if not conta_azul_customer_id:
        conta_azul_customer_id = transaction_data.get("pessoa_id") or transaction_data.get("customer_id")

    # Find client — if no match, still sync but without client link
    client = None
    if conta_azul_customer_id:
        client = db.query(Client).filter_by(
            conta_azul_id=str(conta_azul_customer_id)
        ).first()

    # Check if record already exists
    existing_record = db.query(FinancialRecord).filter_by(
        conta_azul_id=str(conta_azul_id)
    ).first()

    # Parse dates (v2 API field names)
    due_date = _parse_date(
        transaction_data.get("data_vencimento")
        or transaction_data.get("due_date")
    )
    transaction_date = _parse_date(
        transaction_data.get("data_emissao")
        or transaction_data.get("data_competencia")
        or transaction_data.get("created_at")
    )
    payment_date = _parse_date(
        transaction_data.get("data_pagamento")
        or transaction_data.get("paid_at")
    )

    # Calculate days overdue
    days_overdue = 0
    if due_date and not payment_date:
        today = date.today()
        if today > due_date:
            days_overdue = (today - due_date).days

    # Map status (Conta Azul v2 uses uppercase PT-BR)
    raw_status = (transaction_data.get("status", "") or "").upper()
    if raw_status in ("QUITADO", "RECEBIDO", "PAGO", "LIQUIDADO"):
        status = "paid"
    elif raw_status == "CANCELADO":
        status = "cancelled"
    elif days_overdue > 0:
        status = "overdue"
    else:
        status = "pending"

    # Parse amount (v2 field names)
    raw_amount = (
        transaction_data.get("valor")
        or transaction_data.get("valor_total")
        or transaction_data.get("amount")
        or 0
    )
    if isinstance(raw_amount, (int, float)):
        amount = Decimal(str(raw_amount))
    elif isinstance(raw_amount, str):
        amount = Decimal(raw_amount)
    else:
        amount = Decimal("0")

    # Description (v2 field names)
    description = (
        transaction_data.get("descricao")
        or transaction_data.get("description")
        or transaction_data.get("observacao")
        or ""
    )

    # Transaction type from _sync_type tag added during fetch
    sync_type = transaction_data.get("_sync_type", "receivable")
    tx_type = "receivable" if sync_type == "receivable" else "payable"

    # Person name for notes
    nome_pessoa = ""
    if isinstance(pessoa, dict):
        nome_pessoa = pessoa.get("nome", "")

    # Prepare record data
    record_data = {
        "transaction_type": tx_type,
        "transaction_number": str(transaction_data.get("numero", f"CA-{conta_azul_id}")),
        "description": description[:500] if description else "",
        "amount": amount,
        "currency": "BRL",
        "status": status,
        "transaction_date": transaction_date or date.today(),
        "due_date": due_date,
        "payment_date": payment_date,
        "days_overdue": days_overdue,
        "conta_azul_id": str(conta_azul_id),
        "payment_method": transaction_data.get("forma_pagamento"),
        "notes": nome_pessoa[:200] if nome_pessoa else None,
        "last_synced_at": datetime.utcnow()
    }

    # Only set client_id if we found a matching client

        record_data["client_id"] = client.id

    if existing_record:
        for key, value in record_data.items():
            setattr(existing_record, key, value)
        logger.debug(f"Updated financial record: {existing_record.transaction_number}")
        return (False, True)
    else:
        new_record = FinancialRecord(**record_data)
        db.add(new_record)
        logger.debug(f"Created financial record: {new_record.transaction_number}")
        return (True, False)"""
