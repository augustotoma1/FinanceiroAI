new_logger = """logger = logging.getLogger(__name__)

# ── OAuth State Store (CSRF) ──────────────────────────────────────
# In-memory store for OAuth state tokens with TTL.
# Each entry: {state_token: timestamp_created}
# States expire after 10 minutes.
_oauth_states: Dict[str, float] = {}
_STATE_TTL_SECONDS = 600  # 10 minutos

def _store_state(state: str) -> None:
    \"\"\"Store a state token with current timestamp.\"\"\"
    _cleanup_expired_states()
    _oauth_states[state] = time.time()

def _validate_state(state: Optional[str]) -> bool:
    \"\"\"Validate and consume a state token. Returns True if valid.\"\"\"
    _cleanup_expired_states()
    if not state:
        return False
    if state not in _oauth_states:
        return False
    # Consume the state (one-time use)
    del _oauth_states[state]
    return True

def _cleanup_expired_states() -> None:
    \"\"\"Remove expired state tokens.\"\"\"
    now = time.time()
    expired = [s for s, t in _oauth_states.items() if now - t > _STATE_TTL_SECONDS]
    for s in expired:
        del _oauth_states[s]

# Create router with prefix and tags
router = APIRouter()"""

    content = content.replace(old_logger, new_logger, 1)
