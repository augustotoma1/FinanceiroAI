import os, sys, shutil, subprocess, zlib

os.chdir('/opt/agent-financeiro-aisatec')

# Read and decompress hex data
with open('/tmp/hex_data.txt') as f:
    hex_data = f.read().strip().replace('\n', '')

print(f'[1] Hex data: {len(hex_data)} chars')

try:
    data = zlib.decompress(bytes.fromhex(hex_data))
    print(f'[2] Decompressed: {len(data)} bytes')
except Exception as e:
    print(f'[ERR] Decompress failed: {e}')
    sys.exit(1)

if len(data) != 38719:
    print(f'[WARN] Size: got {len(data)}, expected 38719')

F = 'app/services/telegram_bot.py'
shutil.copy2(F, F + '.bak4')
print('[3] Backup -> .bak4')

with open(F, 'wb') as fh:
    fh.write(data)
print('[4] New file written')

r = subprocess.run(
    ['venv/bin/python3', '-c',
     'import py_compile; py_compile.compile("app/services/telegram_bot.py", doraise=True)'],
    capture_output=True, text=True
)
if r.returncode:
    print(f'[ERR] Syntax: {r.stderr}')
    shutil.copy2(F + '.bak4', F)
    print('[RESTORED] Backup restored')
    sys.exit(1)
print('[5] Syntax OK')

# Quick verify key functions exist
with open(F) as fh:
    content = fh.read()
